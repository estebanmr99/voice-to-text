    **1. Expand `src/glossary.py`** — Add import and export methods:

    ```python
    class GlossaryStore:
        # ... existing methods ...

        def import_glossary(self, source_path: Path) -> list[str]:
            """Import a glossary JSON file as the user glossary.

            Validates the file, then copies it to the user glossary location.
            Returns empty list on success, list of error strings on validation failure.
            The imported entries override defaults on conflict when load() is called.
            """

        def export_glossary(self, dest_path: Path) -> None:
            """Export the current merged glossary (defaults + user entries) to a JSON file.

            Creates parent directories if needed. Writes in the same schema as
            default_glossary.json for round-trip compatibility.
            """

        def get_entries_as_dicts(self) -> list[dict]:
            """Return current merged entries as a list of dicts for export.
            Each dict has 'input', 'output', and optional 'context' keys.
            """
    ```

    Key behaviors for `import_glossary()`:
    - Read the source JSON file
    - Validate using existing `validate()` method
    - If validation fails, return the error list (do NOT overwrite existing user glossary)
    - If validation passes, write the entries to `self._user_path` as JSON
    - Return empty list on success
    - If source file doesn't exist or is unreadable, return `["File not found: {source_path}"]`
    - If source JSON is invalid (parse error), return `["Invalid JSON in glossary file"]`

    Key behaviors for `export_glossary()`:
    - Call `get_entries_as_dicts()` to get merged entries
    - Write to `dest_path` as indented JSON with the same schema as `data/default_glossary.json`: `{"entries": [...], "version": 1, "description": "..."}`
    - Create parent directories if they don't exist
    - Raise `OSError` only if write truly fails (disk full, permissions)

    Key behaviors for `get_entries_as_dicts()`:
    - Call `self.load()` to get merged entries
    - Convert each `GlossaryEntry` to `{"input": ..., "output": ..., "context": ...}`
    - Skip entries where `context` is empty (omit the key entirely for cleaner export)

    ```python
    def _on_import_glossary(self) -> None:
        """Import a glossary JSON file and set it as user glossary."""
        path, _ = QFileDialog.getOpenFileName(
            self,
            "Import glossary file",
            "",
            "Glossary files (*.json *.yaml *.yml *.txt)",
        )
        if not path:
            return

        if self._glossary_store is None:
            QMessageBox.warning(self, "Import Error", "Glossary store not initialized.")
            return

        errors = self._glossary_store.import_glossary(Path(path))
        if errors:
            QMessageBox.warning(
                self,
                "Import Error",
                "Invalid glossary file:\n\n" + "\n".join(errors),
            )
            return

        # Update glossary path setting to point to the imported file location
        self._settings.set("glossary_path", str(self._glossary_store._user_path))
        QMessageBox.information(self, "Import Complete", "Glossary imported successfully.")

    def _on_export_glossary(self) -> None:
        """Export current merged glossary to a file."""
        path, _ = QFileDialog.getSaveFileName(
            self,
            "Export glossary file",
            "",
            "Glossary files (*.json)",
        )
        if not path:
            return

        if self._glossary_store is None:
            QMessageBox.warning(self, "Export Error", "Glossary store not initialized.")
            return

        try:
            self._glossary_store.export_glossary(Path(path))
            QMessageBox.information(self, "Export Complete", f"Glossary exported to:\n{path}")
        except OSError as exc:
            QMessageBox.warning(self, "Export Error", f"Failed to write glossary:\n{exc}")
    ```
